\section{Current Position}

\textbf{Research Fellow in Remote Sensing and Modelling for Crop Sustainability and Resilience}, School of Earth and Environment, University of Leeds \hfill Jul 2024 -- Present

\vspace{0.10cm}
\begin{highlights}
    \item Member of the Institute for Climate and Atmospheric Science, contributing to \textbf{iSPARK} (Innovation in Sustainability, Policy, Adaptation and Resilience in Kenya).
    \item Develop multi-sensor Earth observation methods, combining Sentinel-1 and Sentinel-2 time series with machine learning, to monitor cropping systems, vegetation dynamics and agricultural resilience under climate variability in smallholder farming systems across Eastern Africa.
    \item Mapping drought stress in maize monocrop versus intercrop systems in western Kenya to inform climate-resilience policy; presented at the 3rd International Crop Modelling Symposium (iCROPM2026), Florence, Feb 2026.
\end{highlights}
